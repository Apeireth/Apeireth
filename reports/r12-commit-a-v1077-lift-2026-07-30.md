def verify(self, modules):
    try:
        from apeireth.r11_v04_test_ownership import find_tests_owning_module as _r11_find
        r11_available = True
    except Exception:
        r11_available = False  # fallback to filename only

    for info in modules:
        test_path = self.test_dir / f"test_{info.module_name}.py"
        if test_path.exists():
            info.has_test_file = True
            continue
        if r11_available:  # AST fallback
            owners = _r11_find(info.module_name, ...)
            if owners:
                info.has_test_file = True
