```rust
pub fn chat(input: &str) -> String {
    CYCLE_COUNT.fetch_add(1, Ordering::Relaxed);

    // 1. R19 认知循环 (R11 LOCKED 链路, 真后端全接)
    let cycle_target = ActionTarget::NormalAction(format!("tui-chat:{input}"));
    let cognitive = CognitiveInput::new(vec![cycle_target], "tui-chat");
    let cycle_result = run_cycle(cognitive);

    // 2. 真调 LLM (R17 apeireth-api 走 minimaxi OpenAI 协议)
    let llm_result = call_llm_sync(input);

    // 3. 拼装
    match (&cycle_result, &llm_result) {
        (Ok(c), Ok(reply)) => format!(
            "{}\n\n─── R19 认知循环 ───\nv0.5 transferability={:.3} · verdicts={} · cycle#{}",
            reply.text, c.v05.transferability, c.verdicts.len(), CYCLE_COUNT.load(Ordering::Relaxed)
        ),
        (Ok(c), Err(e)) => format!(
            "(LLM 调用失败: {})\n\n─── R19 认知循环 fallback ───\nv0.5 transferability={:.3} · verdicts={}",
            e, c.v05.transferability, c.verdicts.len()
        ),
        // ... 其他 2 个分支
    }
}
```

### 3. call_llm_sync() 内部

```rust
fn call_llm_sync(input: &str) -> Result<LlmReply, String> {
    // 读 apikey (主人在 openclaw/apikey.txt)
    let key = std::fs::read_to_string(APIKEY_PATH)
        .map_err(|e| format!("read apikey.txt: {e}"))?
        .trim()
        .to_string();

    // OpenAI provider
    let cfg = OpenAiCompatibleConfig::new(
        "minimaxi",
        "https://api.minimaxi.com/v1",
        key,
        vec!["MiniMax-M3".to_string()],
    );
    let provider = OpenAiCompatibleProvider::new(cfg)?;

    // request
    let req = LlmRequest {
        model: "MiniMax-M3".into(),
        messages: vec![
            ChatMessage::system("你是 Apeireth 主 AI,基于 R19 立体架构..."),
            ChatMessage::user(input.into()),
        ],
        temperature: 0.7,
        max_tokens: 1024,
        trace_id: None,
        stop: vec![],
    };

    // 同步等 async (main loop 是同步的, 内部建 tokio runtime)
    let rt = tokio::runtime::Builder::new_current_thread()
        .enable_all()
        .build()?;
    let resp = rt.block_on(provider.complete(req))?;
    Ok(LlmReply { text: resp.content, usage: ... })
}
```
