```yaml
- name: Parse run_url to run_id (workflow_dispatch 重签模式)

  id: parse_run
  shell: bash
  run: |
    set -euo pipefail
    RUN_URL="${{ inputs.run_url }}"
    # 期望格式: https://github.com/<owner>/<repo>/actions/runs/<run_id>
    if [[ "${RUN_URL}" =~ /actions/runs/([0-9]+) ]]; then
      RUN_ID="${BASH_REMATCH[1]}"
      echo ">>> run_url: ${RUN_URL} → run_id: ${RUN_ID}"
      echo "run_id=${RUN_ID}" >> "$GITHUB_OUTPUT"
    else
      echo "❌ run_url 格式不合法: ${RUN_URL}"
      echo "   期望: https://github.com/<owner>/<repo>/actions/runs/<id>"
      exit 1
    fi
```

### 3.2 8 包 cosign verify-blob 守门 (verify job 内)

```bash
# verify job step 6/9 (per §1.2)
for sig in dist/signatures/*.sig; do
  [[ -e "${sig}" ]] || continue
  pkg="${sig%.sig}"
  if [[ ! -f "${pkg}" ]]; then
    echo "  ⚠️  跳过: ${pkg} 缺失 (其他 OS build 的包, 此 runner 不可用)"
    continue
  fi
  TOTAL=$((TOTAL+1))
  if cosign verify-blob --key "${COSIGN_PUB}" --signature "${sig}" "${pkg}" >/dev/null 2>&1; then
    echo "  ✓ $(basename ${pkg})"
  else
    echo "  ❌ $(basename ${pkg}) verify 失败"
    FAIL=$((FAIL+1))
  fi
done
```
