### 3.1 公共 API

```rust
// 6 个操作 (per Playwright MCP server schema 1:1 借鉴, snake_case)
pub enum BrowserAction { Navigate, Click, Type, Screenshot, GetText, Close }

// 5 个 params struct (1:1 对应 5 non-trivial action, close = unit)
pub struct BrowserNavigateParams { url, timeout_ms }   // default timeout 30s
pub struct BrowserClickParams { selector, timeout_ms }   // default timeout 5s
pub struct BrowserTypeParams { selector, text, delay_ms } // default delay 0
pub struct BrowserScreenshotParams { format, full_page, filename } // default png
pub struct BrowserGetTextParams { selector }
pub struct BrowserCloseParams {}  // unit, default

// envelope: MCP tools/call arguments 解析目标
pub struct BrowserRequest { action: String, params: Value }
impl BrowserRequest { pub fn from_value(v: &Value) -> Result<Self, String> }

// 4 错误码 (per MCP server-defined -32000~-32099)
pub const BROWSER_INVALID_ACTION:  i32 = -32020
pub const BROWSER_INVALID_PARAMS:  i32 = -32021
pub const BROWSER_CALL_FAILED:     i32 = -32022
pub const BROWSER_NOT_INSTALLED:   i32 = -32023

// handler skeleton (0 真接, R124+ 真接)
pub async fn browser_tool_handler(action: BrowserAction, params: Value) -> Result<Value, String>

// (ToolDef, ToolHandler) 一对, 跟 McpServer::register_tool 配

pub fn browser_tool_handler_fn() -> ToolHandler
```
